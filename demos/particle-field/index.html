Canvas Particle Field

How it works

function loop(now) {
  const fps = 1000 / (now - last);
  last = now;
  ctx.clearRect(0, 0, w, h);
  for (const p of particles) { p.step(); p.draw(ctx); }
  drawLinks(ctx, particles);   // connect nearby particles
  raf = requestAnimationFrame(loop);
}
